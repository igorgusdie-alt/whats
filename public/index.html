const express = require('express'); const { createClient } = require('@supabase/supabase-js'); const path = require('path'); const app = express(); app.use(express.json()); app.use(express.static(path.join(__dirname, 'public'))); // Inicialização do cliente Supabase usando variáveis de ambiente const supabaseUrl = process.env.SUPABASE_URL; const supabaseKey = process.env.SUPABASE_SERVICE_ROLE_KEY || process.env.SUPABASE_ANON_KEY; const supabase = createClient(supabaseUrl, supabaseKey); // ROTA DE LOGIN COMPATÍVEL COM A TABELA USERS app.post('/api/auth/login', async (req, res) => { try { const { email, password } = req.body; if (!email || !password) { return res.status(400).json({ error: 'E-mail e senha são obrigatórios' }); } // Busca o usuário na tabela customizada 'users' const { data: user, error } = await supabase .from('users') .select('*') .eq('email', email) .eq('is_active', true) .single(); if (error || !user) { return res.status(401).json({ error: 'E-mail ou senha inválidos' }); } // Validação da senha if (user.password_hash !== password) { return res.status(401).json({ error: 'E-mail ou senha inválidos' }); } // Retorna os dados do atendente para o frontend res.json({ success: true, agent: { id: user.id, name: user.name, email: user.email, departments: user.departments || ['suporte', 'financeiro', 'comercial'] } }); } catch (err) { console.error('Erro no login:', err); res.status(500).json({ error: 'Erro interno no servidor' }); } }); // ROTA PARA LISTAR TICKETS app.get('/api/tickets', async (req, res) => { try { const { agentId, department } = req.query; let query = supabase.from('tickets').select('*, contacts(*), agents(name)'); if (department) { query = query.eq('department', department); } const { data, error } = await query; if (error) throw error; res.json(data); } catch (err) { console.error('Erro ao buscar tickets:', err); res.status(500).json({ error: 'Erro ao buscar tickets' }); } }); // ROTA PARA BUSCAR MENSAGENS DE UM TICKET app.get('/api/tickets/:id/messages', async (req, res) => { try { const ticketId = req.params.id; const { data, error } = await supabase .from('messages') .select('*') .eq('ticket_id', ticketId) .order('created_at', { ascending: true }); if (error) throw error; res.json(data); } catch (err) { console.error('Erro ao buscar mensagens:', err); res.status(500).json({ error: 'Erro ao buscar mensagens' }); } }); // ROTA PARA ENVIAR MENSAGEM app.post('/api/messages/send', async (req, res) => { try { const { ticketId, phone, message, agentName, agentId } = req.body; const { data, error } = await supabase .from('messages') .insert([{ ticket_id: ticketId, content: message, sender_type: 'agent', sender_name: agentName || 'Atendente' }]); if (error) throw error; res.json({ success: true, data }); } catch (err) { console.error('Erro ao enviar mensagem:', err); res.status(500).json({ error: 'Erro ao enviar mensagem' }); } }); // ROTA PARA LISTAR ATENDENTES app.get('/api/agents', async (req, res) => { try { const { data, error } = await supabase .from('users') .select('id, name, email, departments, is_active') .eq('is_active', true); if (error) throw error; res.json(data); } catch (err) { console.error('Erro ao listar atendentes:', err); res.status(500).json({ error: 'Erro ao listar atendentes' }); } }); // ROTA PARA ASSUMIR OU TRANSFERIR TICKET app.post('/api/tickets/:id/assign', async (req, res) => { try { const ticketId = req.params.id; const { agentId } = req.body; const { data, error } = await supabase .from('tickets') .update({ assigned_to: agentId, status: 'em_atendimento' }) .eq('id', ticketId); if (error) throw error; res.json({ success: true, data }); } catch (err) { console.error('Erro ao atribuir ticket:', err); res.status(500).json({ error: 'Erro ao atribuir ticket' }); } }); // ROTA PARA ALTERAR SETOR app.post('/api/tickets/:id/department', async (req, res) => { try { const ticketId = req.params.id; const { department } = req.body; const { data, error } = await supabase .from('tickets') .update({ department }) .eq('id', ticketId); if (error) throw error; res.json({ success: true, data }); } catch (err) { console.error('Erro ao alterar setor:', err); res.status(500).json({ error: 'Erro ao alterar setor' }); } }); // ROTA PARA ENCERRAR ATENDIMENTO app.post('/api/tickets/:id/close', async (req, res) => { try { const ticketId = req.params.id; const { data, error } = await supabase .from('tickets') .update({ status: 'encerrado' }) .eq('id', ticketId); if (error) throw error; res.json({ success: true, data }); } catch (err) { console.error('Erro ao encerrar atendimento:', err); res.status(500).json({ error: 'Erro ao encerrar atendimento' }); } }); // ROTA PARA ATUALIZAR SENHA DE ATENDENTE app.put('/api/agents/:id/password', async (req, res) => { try { const agentId = req.params.id; const { password } = req.body; if (!password) { return res.status(400).json({ error: 'A nova senha é obrigatória' }); } const { data, error } = await supabase .from('users') .update({ password_hash: password }) .eq('id', agentId); if (error) throw error; res.json({ success: true, message: 'Senha atualizada com sucesso' }); } catch (err) { console.error('Erro ao atualizar senha:', err); res.status(500).json({ error: 'Erro ao atualizar senha' }); } }); // ROTA PARA ATUALIZAR SETORES DE ATENDENTE app.post('/api/agents/:id/departments', async (req, res) => { try { const agentId = req.params.id; const { departments } = req.body; const { data, error } = await supabase .from('users') .update({ departments }) .eq('id', agentId); if (error) throw error; res.json({ success: true, message: 'Setores atualizados com sucesso' }); } catch (err) { console.error('Erro ao atualizar setores:', err); res.status(500).json({ error: 'Erro ao atualizar setores' }); } }); const PORT = process.env.PORT || 3000; app.listen(PORT, () => { console.log(`Servidor rodando na porta ${PORT}`); });
